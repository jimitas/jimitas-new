## アプリデータの設計

アプリを追加するときは `src/data/apps.ts` に1件追加するだけでよい。

```typescript
// src/types/index.ts
export type AppType = "app" | "print" | "tool"

export type AppItem = {
  id: string
  title: string
  description: string
  grades: (1 | 2 | 3 | 4 | 5 | 6 | "中学" | "高校")[]
  subjects: string[]
  tags: string[]
  type: AppType
  path: string
}
```

## ゲーミフィケーション設計

```typescript
// src/hooks/useCoins.ts
// 当面はlocalStorageで管理
// 将来的にSupabaseに移行する

export function useCoins() {
  const getCoins = (): number => {
    return parseInt(localStorage.getItem("jimitas_coins") || "0")
  }
  const addCoins = (amount: number) => {
    const current = getCoins()
    localStorage.setItem("jimitas_coins", String(current + amount))
  }
  return { getCoins, addCoins }
}
```
